cmake_minimum_required(VERSION 3.20)
project(market_data_pipeline VERSION 1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(PIPELINE_SANITIZER "none" CACHE STRING "none, address, or thread")
set_property(CACHE PIPELINE_SANITIZER PROPERTY STRINGS none address thread)
if(NOT PIPELINE_SANITIZER MATCHES "^(none|address|thread)$")
  message(FATAL_ERROR "Unknown PIPELINE_SANITIZER")
endif()
find_package(Threads REQUIRED)
foreach(target pipeline_demo pipeline_benchmark tests_core tests_transport tests_ingress)
  if(target STREQUAL "pipeline_demo")
    set(source demo.cpp)
  elseif(target STREQUAL "pipeline_benchmark")
    set(source benchmark.cpp)
  else()
    set(source ${target}.cpp)
  endif()
  add_executable(${target} ${source})
  target_include_directories(${target} PRIVATE include ../16_cpp_order_book_replay/include)
  target_link_libraries(${target} PRIVATE Threads::Threads)
  if(WIN32)
    target_link_libraries(${target} PRIVATE ws2_32)
  endif()
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /permissive- /EHsc)
    if(NOT PIPELINE_SANITIZER STREQUAL "none")
      message(FATAL_ERROR "Sanitizer configuration requires Linux GCC or Clang")
    endif()
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Wconversion -fno-omit-frame-pointer)
    if(PIPELINE_SANITIZER STREQUAL "address")
      target_compile_options(${target} PRIVATE -fsanitize=address,undefined)
      target_link_options(${target} PRIVATE -fsanitize=address,undefined)
    elseif(PIPELINE_SANITIZER STREQUAL "thread")
      target_compile_options(${target} PRIVATE -fsanitize=thread)
      target_link_options(${target} PRIVATE -fsanitize=thread)
    endif()
  endif()
endforeach()
enable_testing()
# GCC TSan can fail before main with modern Linux address randomization. Apply
# the process-local layout directly to each test binary: wrapping CTest itself
# is insufficient when its launcher (for example Snap) changes personality.
set(PIPELINE_TEST_LAUNCHER)
if(PIPELINE_SANITIZER STREQUAL "thread" AND CMAKE_SYSTEM_NAME STREQUAL "Linux")
  find_program(PIPELINE_SETARCH setarch REQUIRED)
  set(PIPELINE_TEST_LAUNCHER ${PIPELINE_SETARCH} ${CMAKE_SYSTEM_PROCESSOR} -R)
endif()
add_test(NAME protocol_recovery_risk COMMAND ${PIPELINE_TEST_LAUNCHER} $<TARGET_FILE:tests_core>)
add_test(NAME queues_udp COMMAND ${PIPELINE_TEST_LAUNCHER} $<TARGET_FILE:tests_transport>)
add_test(NAME ingress_loss_fence COMMAND ${PIPELINE_TEST_LAUNCHER} $<TARGET_FILE:tests_ingress>)
add_test(NAME loopback_recovery_demo COMMAND ${PIPELINE_TEST_LAUNCHER} $<TARGET_FILE:pipeline_demo>)
add_test(NAME benchmark_smoke COMMAND ${PIPELINE_TEST_LAUNCHER} $<TARGET_FILE:pipeline_benchmark> --events 2000 --repeats 1)
set_tests_properties(protocol_recovery_risk queues_udp ingress_loss_fence loopback_recovery_demo benchmark_smoke PROPERTIES TIMEOUT 60)
