cmake_minimum_required(VERSION 3.20)
project(order_book_replay VERSION 1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(BOOK_SANITIZERS "Enable address and undefined-behavior sanitizers (GCC/Clang)" OFF)
foreach(target book_tests book_benchmark book_replay)
  string(REPLACE "book_" "" source_name ${target})
  add_executable(${target} ${source_name}.cpp)
  target_include_directories(${target} PRIVATE include)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /permissive- /EHsc)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Wconversion)
    if(BOOK_SANITIZERS)
      target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
      target_link_options(${target} PRIVATE -fsanitize=address,undefined)
    endif()
  endif()
endforeach()
enable_testing()
add_test(NAME matching_and_differential COMMAND book_tests)
add_test(NAME replay_fixture COMMAND book_replay ${CMAKE_CURRENT_SOURCE_DIR}/data/example.csv)
set_tests_properties(replay_fixture PROPERTIES PASS_REGULAR_EXPRESSION "events=7 trades=3 resting=1")
file(GENERATE OUTPUT ${CMAKE_CURRENT_BINARY_DIR}/invalid_zero.csv CONTENT "A,1,B,100,0\n")
file(GENERATE OUTPUT ${CMAKE_CURRENT_BINARY_DIR}/invalid_duplicate.csv CONTENT "A,1,B,100,5\nA,1,S,100,5\n")
file(GENERATE OUTPUT ${CMAKE_CURRENT_BINARY_DIR}/invalid_negative.csv CONTENT "A,1,B,100,-1\n")
file(GENERATE OUTPUT ${CMAKE_CURRENT_BINARY_DIR}/invalid_overflow.csv CONTENT "A,1,B,9223372036854775808,1\n")
file(GENERATE OUTPUT ${CMAKE_CURRENT_BINARY_DIR}/invalid_columns.csv CONTENT "A,1,B,100,1,\n")
foreach(case zero duplicate negative overflow columns)
  add_test(NAME replay_rejects_${case} COMMAND book_replay ${CMAKE_CURRENT_BINARY_DIR}/invalid_${case}.csv)
  set_tests_properties(replay_rejects_${case} PROPERTIES WILL_FAIL TRUE)
endforeach()
